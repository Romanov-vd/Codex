cmake_minimum_required(VERSION 3.16)
project(Lab2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(image_filters src/Image.cpp src/Filters.cpp)
target_include_directories(image_filters PUBLIC include)
add_executable(Lab src/Source.cpp)
target_link_libraries(Lab PRIVATE image_filters)
add_executable(Tests tests/tests_image.cpp tests/tests_filters.cpp)
target_link_libraries(Tests PRIVATE image_filters)
if(MSVC)
    target_compile_options(image_filters PRIVATE /utf-8)
    target_compile_options(Lab PRIVATE /utf-8)
    target_compile_options(Tests PRIVATE /utf-8)
    target_compile_definitions(image_filters PRIVATE _CRT_SECURE_NO_WARNINGS)
    set_property(TARGET Lab Tests PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
endif()
enable_testing()
add_test(NAME ImageAndFilters COMMAND Tests)
